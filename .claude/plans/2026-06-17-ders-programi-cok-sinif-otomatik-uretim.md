```csharp
public static IReadOnlyList<LessonDemand> Build(
    Guid branchId, IReadOnlyList<AssignmentLine> lines, IReadOnlySet<Guid> hardSubjectIds)

    foreach (var line in lines)

    return demands;
```

```csharp
public sealed record FeasibilityContext(
    IReadOnlySet<(Guid BranchId, DayOfWeek Day, int Period)> ClassOccupied,
    IReadOnlySet<(Guid TeacherId, DayOfWeek Day, int Period)> TeacherOccupied,
    IReadOnlySet<(Guid RoomId, DayOfWeek Day, int Period)> RoomOccupied,
    IReadOnlySet<(Guid TeacherId, DayOfWeek Day, int Period)> ExternalTeacherBusy,
    IReadOnlySet<(Guid RoomId, DayOfWeek Day, int Period)> ExternalRoomBusy,
    IReadOnlyDictionary<Guid, IReadOnlySet<TimeSlot>> TeacherBlockedSlots,
    IReadOnlyDictionary<Guid, Guid?> HomeRoomByBranch,
    IReadOnlyDictionary<(Guid BranchId, DayOfWeek Day, Guid SubjectId), int> DailySubjectCount,
    bool LimitDailySameSubject);

public static class SlotFeasibility
{
    public static bool CanPlace(LessonDemand demand, TimeSlot slot, FeasibilityContext ctx)
    {
        // 1) sınıf aynı slota iki ders alamaz (branch'e özel)
        if (ctx.ClassOccupied.Contains((demand.BranchId, slot.Day, slot.Period))) return false;
        // 2) öğretmen dış programlarda meşgul (global)
        if (ctx.ExternalTeacherBusy.Contains((demand.TeacherId, slot.Day, slot.Period))) return false;
        // 3) öğretmen bu üretimde zaten başka derste (global — sınıflar arası dahil)
        if (ctx.TeacherOccupied.Contains((demand.TeacherId, slot.Day, slot.Period))) return false;
        // 4) öğretmen müsaitlik bloğu (Faz 4; no-op)
        if (ctx.TeacherBlockedSlots.TryGetValue(demand.TeacherId, out var blocked) && blocked.Contains(slot)) return false;
        // 5) ev-dersliği meşgul (branch'in odası varsa; global oda doluluğu)
        if (ctx.HomeRoomByBranch.TryGetValue(demand.BranchId, out var maybeRoom) && maybeRoom is Guid room)
        {
            if (ctx.ExternalRoomBusy.Contains((room, slot.Day, slot.Period))) return false;
            if (ctx.RoomOccupied.Contains((room, slot.Day, slot.Period))) return false;
        }
        // 6) günde aynı dersten en fazla 2 (branch'e özel sayaç)
        if (ctx.LimitDailySameSubject
            && ctx.DailySubjectCount.TryGetValue((demand.BranchId, slot.Day, demand.SubjectId), out var dailyCount)
            && dailyCount >= 2)
        {
            return false;
        }
        return true;
    }
}
```

```csharp
public sealed class GreedyState
{
    public HashSet<(Guid BranchId, DayOfWeek Day, int Period)> ClassOccupied { get; init; } = new();
    public HashSet<(Guid TeacherId, DayOfWeek Day, int Period)> TeacherOccupied { get; init; } = new();
    public HashSet<(Guid RoomId, DayOfWeek Day, int Period)> RoomOccupied { get; init; } = new();
    public Dictionary<(Guid BranchId, DayOfWeek Day, Guid SubjectId), int> DailySubjectCount { get; init; } = new();
    public Dictionary<(Guid BranchId, DayOfWeek Day, int Period), Guid> ClassSubjectAt { get; init; } = new();
    public required FeasibilityContext Context { get; init; }

    public static GreedyState Create(
        IReadOnlySet<(Guid TeacherId, DayOfWeek Day, int Period)> externalTeacherBusy,
        IReadOnlySet<(Guid RoomId, DayOfWeek Day, int Period)> externalRoomBusy,
        IReadOnlyDictionary<Guid, IReadOnlySet<TimeSlot>> teacherBlockedSlots,
        IReadOnlyDictionary<Guid, Guid?> homeRoomByBranch,
        bool limitDailySameSubject)
    {
        var classOccupied = new HashSet<(Guid, DayOfWeek, int)>();
        var teacherOccupied = new HashSet<(Guid, DayOfWeek, int)>();
        var roomOccupied = new HashSet<(Guid, DayOfWeek, int)>();
        var dailySubjectCount = new Dictionary<(Guid, DayOfWeek, Guid), int>();

        return new GreedyState
        {
            ClassOccupied = classOccupied,
            TeacherOccupied = teacherOccupied,
            RoomOccupied = roomOccupied,
            DailySubjectCount = dailySubjectCount,
            ClassSubjectAt = new Dictionary<(Guid, DayOfWeek, int), Guid>(),
            Context = new FeasibilityContext(
                ClassOccupied: classOccupied,
                TeacherOccupied: teacherOccupied,
                RoomOccupied: roomOccupied,
                ExternalTeacherBusy: externalTeacherBusy,
                ExternalRoomBusy: externalRoomBusy,
                TeacherBlockedSlots: teacherBlockedSlots,
                HomeRoomByBranch: homeRoomByBranch,
                DailySubjectCount: dailySubjectCount,
                LimitDailySameSubject: limitDailySameSubject),
        };
    }
}
```

`Place` (branch-keyed mutasyon; oda branch'ten):

```csharp
private static void Place(
    GreedyState state, IReadOnlyDictionary<Guid, Guid?> homeRoomByBranch, LessonDemand demand, TimeSlot slot, List<PlannedPlacement> placements)
{
    var homeRoomId = homeRoomByBranch.TryGetValue(demand.BranchId, out var r) ? r : null;
    state.ClassOccupied.Add((demand.BranchId, slot.Day, slot.Period));
    state.TeacherOccupied.Add((demand.TeacherId, slot.Day, slot.Period));
    state.ClassSubjectAt[(demand.BranchId, slot.Day, slot.Period)] = demand.SubjectId;
    var dailyKey = (demand.BranchId, slot.Day, demand.SubjectId);
    state.DailySubjectCount[dailyKey] = state.DailySubjectCount.GetValueOrDefault(dailyKey) + 1;
    if (homeRoomId is Guid room)
        state.RoomOccupied.Add((room, slot.Day, slot.Period));

    placements.Add(new PlannedPlacement(
        demand.BranchId, slot.Day, slot.Period, demand.SubjectId, demand.TeacherId, homeRoomId, IsBlock: false, BlockGroupSeq: 0));
}
```

```csharp
public sealed class AutoGenClassResolver(IApplicationDbContext db) : IAutoGenClassResolver
{
    public async Task<IReadOnlyList<Guid>> ResolveAsync(
        GenerationScope scope, Guid? branchId, int? gradeLevel, Guid academicTermId, CancellationToken ct)
    {
        var sessionId = await db.AcademicSessions.AsNoTracking()
            .Where(s => s.Terms.Any(t => t.Id == academicTermId))
            .Select(s => (Guid?)s.Id).FirstOrDefaultAsync(ct);
        if (sessionId is null) return Array.Empty<Guid>();

        var assignedBranchIds = await db.TeachingAssignments.AsNoTracking()
            .Where(a => a.AcademicSessionId == sessionId && a.RevokedAt == null)
            .Select(a => a.ClassRoomId).Distinct().ToListAsync(ct);
        if (assignedBranchIds.Count == 0) return Array.Empty<Guid>();

        var query = db.ClassRooms.AsNoTracking()
            .Where(c => assignedBranchIds.Contains(c.Id) && c.Status != ClassRoomStatus.Archived);

        query = scope switch
        {
            GenerationScope.Single => query.Where(c => c.Id == branchId),
            GenerationScope.GradeLevel => query.Where(c =>
                db.GradeLevels.Any(g => g.Id == c.GradeLevelId && g.DisplayOrder == gradeLevel)),
            _ => query, // All
        };

        return await query.Select(c => c.Id).ToListAsync(ct);
    }
}
```

```csharp
        // Her şube için görevlendirme → talepler (branch-etiketli); ev dersliği → harita.

        var homeRoomByBranch = new Dictionary<Guid, Guid?>();
        foreach (var branchId in branchIds)
        {
            var lines = await assignments.GetForBranchAsync(schoolId, branchId, job.AcademicTermId, ct);
            allDemands.AddRange(LessonDemandBuilder.Build(branchId, lines, hardSubjectIds: new HashSet<Guid>()));
            var room = await db.ClassRooms.AsNoTracking()
                .Where(c => c.Id == branchId).Select(c => c.RoomId).FirstOrDefaultAsync(ct);
            homeRoomByBranch[branchId] = room == Guid.Empty ? null : room;
        }

        var periodCount = await bells.GetPeriodCountAsync(schoolId, branchIds.FirstOrDefault(), ct);
        var slots = BuildWeekGrid(periodCount);

        // Dış doluluk: KAPSAM DIŞI şubelerin rezerve eden yerleşimleri (kapsam içi joint çözülür).
        var (extTeacher, extRoom) = await GatherExternalOccupancyAsync(job.AcademicTermId, branchIds, ct);

        var teacherIds = allDemands.Select(d => d.TeacherId).Distinct().ToList();
        var blocked = await availability.GetBlockedSlotsAsync(schoolId, job.AcademicTermId, teacherIds, ct);

        var weights = JsonSerializer.Deserialize<SolverWeights>(job.WeightsJson, AutoGenJson.Options) ?? SolverWeights.Default;

        var input = new SolveInput(
            allDemands, slots, extTeacher, extRoom, homeRoomByBranch, weights, job.Strict, CandidateCount, blocked);

        var result = solver.Solve(input);

        if (result.NoSolution)
        {
            var hints = result.RelaxationHints.Select(h => new AutoGenHintDto(h.Kind.ToString(), h.Reason)).ToList();
            job.MarkNoSolution(JsonSerializer.Serialize(hints, AutoGenJson.Options), clock.UtcNow);
        }
        else
        {
            var classNames = await LoadClassNamesAsync(branchIds, ct); // Dictionary<Guid,string>
            var dtos = result.Candidates.Select(c => MapCandidate(c, classNames)).ToList();
            job.CompleteWithCandidates(JsonSerializer.Serialize(dtos, AutoGenJson.Options), clock.UtcNow);
        }

        await db.SaveChangesAsync(ct);
    }
    catch (Exception ex)
    {
        logger.LogError(ex, "Auto-generate job {JobId} başarısız", jobId);
        job.Fail("internal-error", clock.UtcNow);
        await db.SaveChangesAsync(ct);
    }
}
```

```csharp
public sealed class ApplyAutoGenerateDraftsCommandHandler(
    IApplicationDbContext db, ITenantContext tenant, IScheduleProgramStatsRecomputer stats)
    : ICommandHandler<ApplyAutoGenerateDraftsCommand, IReadOnlyList<AppliedDraftDto>>
{
    public async Task<Result<IReadOnlyList<AppliedDraftDto>>> Handle(
        ApplyAutoGenerateDraftsCommand request, CancellationToken cancellationToken)
    {
        if (tenant.CurrentSchoolId is not { } schoolId)
            return Result<IReadOnlyList<AppliedDraftDto>>.Forbidden();

        var job = await db.ScheduleGenerationJobs.AsNoTracking()
            .FirstOrDefaultAsync(j => j.Id == request.JobId, cancellationToken);
        if (job is null) return Result<IReadOnlyList<AppliedDraftDto>>.NotFound();
        if (job.Status != GenerationStatus.Done)
            return Result<IReadOnlyList<AppliedDraftDto>>.Failure(new Error(
                "timetable.autogen.not-applicable", "Yalnızca tamamlanmış bir işin adayı uygulanabilir."));

        var candidates = job.CandidatesJson is not null
            ? JsonSerializer.Deserialize<IReadOnlyList<AutoGenCandidateDto>>(job.CandidatesJson, AutoGenJson.Options)
            : null;
        if (candidates is null || candidates.Count == 0)
            return Result<IReadOnlyList<AppliedDraftDto>>.Failure(new Error(
                "timetable.autogen.not-applicable", "Uygulanabilir aday yok."));

        var candidate = request.CandidateId is { } cid
            ? candidates.FirstOrDefault(c => c.Id == cid)
            : candidates.FirstOrDefault(c => c.IsRecommended) ?? candidates[0];
        if (candidate is null) return Result<IReadOnlyList<AppliedDraftDto>>.NotFound();

        var applied = new List<AppliedDraftDto>(request.BranchIds.Count);
        foreach (var branchId in request.BranchIds.Distinct())
        {
            // Idempotency: bu job+branch için zaten Taslak varsa onu döndür.
            var existing = await db.SchedulePrograms
                .Where(p => p.GeneratedFromJobId == request.JobId && p.BranchId == branchId)
                .Select(p => p.Id).FirstOrDefaultAsync(cancellationToken);
            if (existing != Guid.Empty) { applied.Add(new AppliedDraftDto(branchId, existing)); continue; }

            var branchPlacements = candidate.Placements.Where(p => p.BranchId == branchId).ToList();
            if (branchPlacements.Count == 0) continue; // bu sınıf için yerleşim yok → atla

            var snapshot = branchPlacements.Select(p => new RestorePlacementInput(
                (DayOfWeek)p.Day, p.Period, p.SubjectId, p.TeacherId, p.RoomId,
                IsBlock: false, BlockGroupId: null)).ToList();

            var program = ScheduleProgram.Create(schoolId, job.AcademicYearId, job.AcademicTermId, branchId);
            program.ApplyGeneratedDraft(snapshot);
            program.StampGeneratedFrom(request.JobId);
            db.SchedulePrograms.Add(program);
            await db.SaveChangesAsync(cancellationToken);
            await stats.SaveProgramWithStatsAsync(program, cancellationToken);

            applied.Add(new AppliedDraftDto(branchId, program.Id));
        }

        return Result<IReadOnlyList<AppliedDraftDto>>.Success(applied);
    }
}
```
